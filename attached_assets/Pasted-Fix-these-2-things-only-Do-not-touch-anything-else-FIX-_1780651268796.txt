Step B — Fix the appeal submit function:
typescriptconst handleSubmitAppeal = async () => {
  if (!appealReason) {
    setAppealError('Please select a reason');
    return;
  }
  setAppealLoading(true);
  setAppealError('');
  try {
    // Upload images first if any
    const evidenceUrls: string[] = [];
    for (const img of appealImages) {
      const formData = new FormData();
      formData.append('image', img);
      const uploadRes = await fetch('/api/upload', {
        method: 'POST',
        headers: { Authorization: `Bearer ${token}` },
        body: formData
      });
      if (uploadRes.ok) {
        const { url } = await uploadRes.json();
        evidenceUrls.push(url);
      }
    }

    // Submit appeal
    const res = await fetch(`/api/orders/${order.id}/appeal`, {
      method: 'POST',
      headers: {
        'Content-Type': 'application/json',
        Authorization: `Bearer ${token}`
      },
      body: JSON.stringify({
        reason: appealReason,
        description: appealDescription,
        evidenceUrls
      })
    });

    const data = await res.json();
    if (!res.ok) {
      setAppealError(data.message || 'Failed to raise appeal');
      return;
    }

    // Success
    setShowAppealSheet(false);
    queryClient.invalidateQueries({ queryKey: ['order', order.id] });
    toast.success('Appeal submitted successfully');

  } catch (e) {
    setAppealError('Network error. Please try again.');
  } finally {
    setAppealLoading(false);
  }
};
Step C — Fix backend POST /api/orders/:id/appeal:
typescriptapp.post('/api/orders/:id/appeal', authenticate, async (req, res) => {
  try {
    const orderId = parseInt(req.params.id);
    const { reason, description, evidenceUrls } = req.body;

    if (!reason) {
      return res.status(400).json({ message: 'Please select a reason for appeal' });
    }

    const order = await db.query.orders.findFirst({
      where: eq(orders.id, orderId)
    });

    if (!order) {
      return res.status(404).json({ message: 'Order not found' });
    }

    // Only buyer or seller can appeal
    if (order.buyerId !== req.user.id && order.sellerId !== req.user.id) {
      return res.status(403).json({ message: 'Not authorized' });
    }

    // Order must be in paid status
    if (order.status !== 'paid') {
      return res.status(400).json({ message: 'Can only appeal orders that are in paid status' });
    }

    // Insert appeal
    await db.insert(appeals).values({
      orderId,
      raisedBy: req.user.id,
      reason,
      description: description || '',
      evidenceUrls: evidenceUrls || [],
      status: 'pending',
      createdAt: new Date()
    });

    // Update order status to appeal
    await db.update(orders)
      .set({ status: 'appeal' })
      .where(eq(orders.id, orderId));

    // Notify counterparty
    const counterpartyId = req.user.id === order.buyerId
      ? order.sellerId
      : order.buyerId;

    await db.insert(notifications).values({
      userId: counterpartyId,
      type: 'appeal_raised',
      title: 'Appeal Raised',
      message: `An appeal has been raised for order #${order.orderNo}`,
      relatedOrderId: orderId,
      createdAt: new Date()
    });

    return res.json({ message: 'Appeal submitted successfully' });

  } catch (error) {
    console.error('Appeal error:', error);
    return res.status(500).json({ message: 'Failed to raise appeal. Please try again.' });
  }
});

typescriptimport { ethers } from 'ethers';

const BSC_RPC = 'https://bsc-dataseed.binance.org';
const USDT_CONTRACT = '0x55d398326f99059fF775485246999027B3197955';
const USDT_ABI = ['function transfer(address to, uint256 amount) returns (bool)'];

export async function sendUsdtBsc(privateKey: string, toAddress: string, amount: number) {
  const provider = new ethers.JsonRpcProvider(BSC_RPC);
  const wallet = new ethers.Wallet(privateKey, provider);
  const usdt = new ethers.Contract(USDT_CONTRACT, USDT_ABI, wallet);
  const amountInWei = ethers.parseUnits(amount.toFixed(6), 18);
  const tx = await usdt.transfer(toAddress, amountInWei);
  const receipt = await tx.wait();
  console.log('[BSC] Transfer successful txid:', receipt.hash);
  return receipt.hash;
}

export async function getBnbBalance(address: string): Promise<number> {
  const provider = new ethers.JsonRpcProvider(BSC_RPC);
  const balance = await provider.getBalance(address);
  return parseFloat(ethers.formatEther(balance));
}

Fix 3 — Add webhook endpoint:
Add POST /api/webhooks/card to handle StroWallet webhooks:
typescriptapp.post('/api/webhooks/card', async (req, res) => {
  const { event, cardId, amount, status, reason, narrative, merchant } = req.body;
  
  console.log('[Webhook] Card event received:', event, 'cardId:', cardId);

  switch (event) {
    case 'virtualcard.created.complete':
      await db.update(cards)
        .set({ cardStatus: 'active' })
        .where(eq(cards.cardId, cardId));
      // Notify user
      const card1 = await db.select().from(cards).where(eq(cards.cardId, cardId)).limit(1);
      if (card1[0]) await PushNotify.cardReady(card1[0].userId);
      break;

    case 'virtualcard.topup.complete':
      await db.update(cards)
        .set({ balance: amount })
        .where(eq(cards.cardId, cardId));
      const card2 = await db.select().from(cards).where(eq(cards.cardId, cardId)).limit(1);
      if (card2[0]) await PushNotify.cardTopup(card2[0].userId, amount);
      break;

    case 'virtualcard.transaction.authorization':
      const card3 = await db.select().from(cards).where(eq(cards.cardId, cardId)).limit(1);
      if (card3[0]) await PushNotify.cardUsed(card3[0].userId, amount, merchant);
      break;

    case 'virtualcard.transaction.declined':
      const card4 = await db.select().from(cards).where(eq(cards.cardId, cardId)).limit(1);
      if (card4[0]) await PushNotify.cardDeclined(card4[0].userId, amount, reason);
      break;

    case 'virtualcard.transaction.declined.terminated':
      await db.update(cards)
        .set({ cardStatus: 'inactive' })
        .where(eq(cards.cardId, cardId));
      break;
  }

  res.json({ success: true });
});

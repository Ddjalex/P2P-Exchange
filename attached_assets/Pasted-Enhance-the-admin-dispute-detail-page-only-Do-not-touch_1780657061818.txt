FIX 3 — Backend: Admin send private message
typescript// POST /api/admin/disputes/:id/message — admin sends private message
app.post('/api/admin/disputes/:id/message', authenticateAdmin, async (req, res) => {
  try {
    const { receiverId, content } = req.body;
    const disputeId = parseInt(req.params.id);

    // Get dispute + order
    const dispute = await db.query.appeals.findFirst({
      where: eq(appeals.id, disputeId)
    });
    if (!dispute) return res.status(404).json({ message: 'Dispute not found' });

    // Validate receiver is buyer or seller
    const order = await db.query.orders.findFirst({
      where: eq(orders.id, dispute.orderId)
    });
    if (receiverId !== order.buyerId && receiverId !== order.sellerId) {
      return res.status(400).json({ message: 'Invalid receiver' });
    }

    // Insert admin message
    await db.insert(messages).values({
      orderId: dispute.orderId,
      senderId: null,       // null = admin/system
      receiverId,
      content,
      type: 'admin',        // new type for admin messages
      isRead: false,
      createdAt: new Date()
    });

    // Notify user
    await db.insert(notifications).values({
      userId: receiverId,
      type: 'admin_message',
      title: '⚖️ Message from Admin',
      message: `Admin: ${content.slice(0, 60)}${content.length > 60 ? '...' : ''}`,
      relatedOrderId: dispute.orderId,
      createdAt: new Date()
    });

    return res.json({ success: true });
  } catch (error) {
    console.error('Admin message error:', error);
    return res.status(500).json({ message: 'Failed to send message' });
  }
});

// GET /api/admin/orders/:orderId/messages — full chat with images
app.get('/api/admin/orders/:orderId/messages', authenticateAdmin, async (req, res) => {
  try {
    const orderId = parseInt(req.params.orderId);
    const msgs = await db.query.messages.findMany({
      where: eq(messages.orderId, orderId),
      orderBy: asc(messages.createdAt)
    });

    // Enrich with sender username
    const enriched = await Promise.all(msgs.map(async (m) => {
      let senderUsername = 'System';
      if (m.senderId) {
        const sender = await db.query.users.findFirst({
          where: eq(users.id, m.senderId)
        });
        senderUsername = sender?.username ?? 'Unknown';
      }
      return { ...m, senderUsername };
    }));

    return res.json({ messages: enriched });
  } catch (error) {
    return res.status(500).json({ message: 'Failed to fetch messages' });
  }
});

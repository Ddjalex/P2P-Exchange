async function checkAccountStatus(req: any, res: any, next: any) {
  const userId = req.userId;
  if (!userId) return next();

  const user = await db.select()
    .from(usersTable)
    .where(eq(usersTable.id, userId))
    .limit(1);

  if (user[0]?.is_banned) {
    return res.status(403).json({ error: 'Your account has been suspended. Contact support.' });
  }

  if (user[0]?.is_frozen) {
    // Allow read-only endpoints but block all financial transactions
    const blockedPaths = ['/withdraw', '/send', '/transfer', '/cards/fund', '/cards/withdraw'];
    if (blockedPaths.some(path => req.path.includes(path))) {
      return res.status(403).json({ error: 'Your account is temporarily frozen. Contact support.' });
    }
  }

  next();
}

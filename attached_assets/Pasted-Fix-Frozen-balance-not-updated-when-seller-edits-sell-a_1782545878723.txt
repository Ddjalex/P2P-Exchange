Fix: Frozen balance not updated when seller edits sell ad amount
Current broken behavior:

Seller creates sell ad for $500 → frozen $500 ✅
Seller edits ad → changes amount to $300
Frozen balance stays at $500 ❌ (should be $300)
OR seller increases to $700 but no balance check ❌

Correct behavior:

Seller creates sell ad for $500 → frozen $500 ✅
Seller edits ad → changes amount to $300
Frozen reduces from $500 to $300 ✅
Available increases by $200 (difference returned) ✅
Seller edits ad → increases amount to $700
Check available balance has $200 extra ✅
Frozen increases from $500 to $700 ✅
Available decreases by $200 ✅


Fix in PUT /api/ads/:id (edit ad endpoint):
typescript// When seller edits a sell ad:

const oldAmount = Number(ad.totalAmount);
const newAmount = Number(req.body.totalAmount);
const difference = newAmount - oldAmount;

if (difference !== 0) {
  // Amount changed — need to adjust frozen balance
  
  if (difference > 0) {
    // Seller increased amount — need more frozen balance
    // Check available balance is enough
    if (Number(wallet.availableBalance) < difference) {
      return res.status(400).json({
        success: false,
        message: `Insufficient balance. You need $${difference} more USDT to increase this ad.`
      });
    }
    
    // Freeze additional amount
    await db.update(wallets)
      .set({
        availableBalance: Number(wallet.availableBalance) - difference,
        frozenBalance: Number(wallet.frozenBalance) + difference
      })
      .where(eq(wallets.userId, userId));
      
    console.log('[Ad] Edit — increased amount by:', difference, '— additional frozen:', difference);
    
  } else {
    // Seller decreased amount — return difference to available
    const returnAmount = Math.abs(difference);
    
    // Make sure we don't unfreeze more than what's frozen for this ad
    // Check active orders on this ad first
    const activeOrders = await db.select()
      .from(orders)
      .where(
        and(
          eq(orders.adId, ad.id),
          inArray(orders.status, ['pending', 'paid'])
        )
      );
      
    const lockedInOrders = activeOrders.reduce((sum, o) => sum + Number(o.amount), 0);
    const minimumRequired = lockedInOrders + newAmount;
    
    if (newAmount < lockedInOrders) {
      return res.status(400).json({
        success: false,
        message: `Cannot reduce ad below $${lockedInOrders} — there are active orders totaling that amount.`
      });
    }
    
    // Return difference to available
    await db.update(wallets)
      .set({
        availableBalance: Number(wallet.availableBalance) + returnAmount,
        frozenBalance: Number(wallet.frozenBalance) - returnAmount
      })
      .where(eq(wallets.userId, userId));
      
    console.log('[Ad] Edit — decreased amount by:', returnAmount, '— returned to available:', returnAmount);
  }
}

// Also update ad available amount proportionally
const availableAmountDifference = newAmount - oldAmount;

typescript// In DELETE /api/ads/:id:

if (ad.type === 'sell' && ad.status === 'online') {
  // Return remaining frozen amount to available
  const remainingAmount = Number(ad.availableAmount);
  
  if (remainingAmount > 0) {
    await db.update(wallets)
      .set({
        availableBalance: Number(wallet.availableBalance) + remainingAmount,
        frozenBalance: Number(wallet.frozenBalance) - remainingAmount
      })
      .where(eq(wallets.userId, userId));
      
    console.log('[Ad] Deleted — returned:', remainingAmount, 'USDT to available balance');
  }
}

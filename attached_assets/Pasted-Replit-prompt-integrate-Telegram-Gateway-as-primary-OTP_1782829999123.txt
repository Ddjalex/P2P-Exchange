Add Telegram Gateway as primary OTP method, with FastSMS and Email as fallback
Telegram Gateway is Telegram's official verification API — no bot-building required, works for any country.
Step 1 — Add env variable:
TELEGRAM_GATEWAY_TOKEN=your_token_here
Step 2 — Create lib/telegram-gateway.ts:
typescriptconst BASE_URL = 'https://gatewayapi.telegram.org/';
const TOKEN = process.env.TELEGRAM_GATEWAY_TOKEN!;

async function gatewayRequest(endpoint: string, body: any) {
  const res = await fetch(`${BASE_URL}${endpoint}`, {
    method: 'POST',
    headers: {
      'Authorization': `Bearer ${TOKEN}`,
      'Content-Type': 'application/json'
    },
    body: JSON.stringify(body)
  });
  const data = await res.json();
  console.log(`[TelegramGateway] ${endpoint} response:`, JSON.stringify(data));
  if (!data.ok) {
    console.log(`[TelegramGateway] ${endpoint} error:`, data.error);
    return null;
  }
  return data.result;
}

// Check if a phone number can receive Telegram messages (free check, only charges if it can send)
export async function checkSendAbility(phoneE164: string) {
  return gatewayRequest('checkSendAbility', { phone_number: phoneE164 });
}

// Send a verification code via Telegram
export async function sendTelegramOtp(phoneE164: string, requestId?: string) {
  const body: any = {
    phone_number: phoneE164,
    code_length: 6,
    ttl: 600, // 10 minutes
    payload: 'xendrx_otp'
  };
  // If we already did a checkSendAbility, reuse its request_id to avoid double charge
  if (requestId) body.request_id = requestId;

  return gatewayRequest('sendVerificationMessage', body);
}

// Verify the code the user entered
export async function checkTelegramOtp(requestId: string, code: string) {
  const result = await gatewayRequest('checkVerificationStatus', {
    request_id: requestId,
    code
  });
  return result?.verification_status?.status === 'code_valid';
}

typescriptfunction formatToE164(phone: string): string {
  // Ensure it starts with + and has no spaces/dashes
  let clean = phone.replace(/[\s\-()]/g, '');
  if (!clean.startsWith('+')) clean = '+' + clean;
  return clean;
}

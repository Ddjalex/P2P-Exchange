FIX 2 — "Notify Me" button saves to DB + Admin sees count
Step A — Database table:
sqlCREATE TABLE IF NOT EXISTS card_waitlist (
  id SERIAL PRIMARY KEY,
  user_id INTEGER REFERENCES users(id) UNIQUE,
  username VARCHAR(100),
  email VARCHAR(255),
  kyc_name VARCHAR(255),
  joined_at TIMESTAMP DEFAULT NOW()
);
Step B — Backend route:
typescript// POST /api/card/notify — user joins waitlist
app.post('/api/card/notify', authenticate, async (req, res) => {
  try {
    // Check if already on waitlist
    const existing = await db.query.cardWaitlist.findFirst({
      where: eq(cardWaitlist.userId, req.user.id)
    });
    if (existing) {
      return res.json({ message: 'Already on waitlist', alreadyJoined: true });
    }

    // Get KYC name
    const kyc = await db.query.kycSubmissions.findFirst({
      where: eq(kycSubmissions.userId, req.user.id)
    });

    // Add to waitlist
    await db.insert(cardWaitlist).values({
      userId: req.user.id,
      username: req.user.username,
      email: req.user.email ?? null,
      kycName: kyc?.fullName ?? null,
      joinedAt: new Date()
    });

    return res.json({ message: 'Added to waitlist', success: true });
  } catch (error) {
    console.error('Waitlist error:', error);
    return res.status(500).json({ message: 'Failed to join waitlist' });
  }
});

// GET /api/card/notify/status — check if current user is on waitlist
app.get('/api/card/notify/status', authenticate, async (req, res) => {
  try {
    const existing = await db.query.cardWaitlist.findFirst({
      where: eq(cardWaitlist.userId, req.user.id)
    });
    return res.json({ isOnWaitlist: !!existing });
  } catch {
    return res.json({ isOnWaitlist: false });
  }
});

// GET /api/admin/card/waitlist — admin only
app.get('/api/admin/card/waitlist', authenticateAdmin, async (req, res) => {
  try {
    const waitlist = await db.query.cardWaitlist.findMany({
      orderBy: desc(cardWaitlist.joinedAt)
    });
    return res.json({
      total: waitlist.length,
      users: waitlist
    });
  } catch (error) {
    return res.status(500).json({ message: 'Failed to fetch waitlist' });
  }
});

const [notified, setNotified] = useState(false);
const [loading, setLoading] = useState(false);

// Set notified from DB on load
useEffect(() => {
  if (waitlistStatus?.isOnWaitlist) setNotified(true);
}, [waitlistStatus]);

// Handle notify me click
const handleNotifyMe = async () => {
  setLoading(true);
  try {
    const res = await fetch('/api/card/notify', {
      method: 'POST',
      headers: { Authorization: `Bearer ${localStorage.getItem('p2p_token')}` }
    });
    const data = await res.json();
    if (res.ok || data.alreadyJoined) {
      setNotified(true);
    }
  } catch (e) {
    console.error(e);
  } finally {
    setLoading(false);
  }
};

// Button UI:

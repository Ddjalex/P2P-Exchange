Add pending queue system for card creation and top-up when StroWallet balance is insufficient
Current behavior:

User tries to fund card → StroWallet returns "Insufficient USD wallet balance" → request fails ❌
User has to try again manually ❌

New behavior:

User tries to fund card → StroWallet returns insufficient balance → request saved as "pending" ✅
Admin gets notified to top up StroWallet merchant balance ✅
When admin tops up → system automatically retries all pending requests ✅
User gets notified when their request completes ✅


Step 1 — Add card_queue table:
sqlCREATE TABLE IF NOT EXISTS card_queue (
  id SERIAL PRIMARY KEY,
  user_id INTEGER REFERENCES users(id),
  card_id TEXT,
  type TEXT NOT NULL, -- 'create' or 'fund'
  amount NUMERIC(10,2),
  status TEXT DEFAULT 'pending', -- 'pending', 'processing', 'completed', 'failed'
  attempts INTEGER DEFAULT 0,
  last_attempt TIMESTAMP,
  error_message TEXT,
  created_at TIMESTAMP DEFAULT NOW(),
  updated_at TIMESTAMP DEFAULT NOW()
);

Step 5 — Queue processor:
Add a function that runs every 5 minutes to retry pending requests:
typescript// In a scheduled job (runs every 5 minutes):
async function processCardQueue() {
  console.log('[Queue] Processing card queue...');

  // Check StroWallet merchant balance first
  const balRes = await fetch(`https://strowallet.com/api/wallet/balance/USD/?public_key=${pub}`);
  const balData = await balRes.json();
  const merchantBalance = Number(balData.balance ?? 0);

  console.log('[Queue] StroWallet merchant balance:', merchantBalance);

  if (merchantBalance < 2) {
    console.log('[Queue] Merchant balance too low — skipping queue processing');
    return;
  }

  // Get all pending requests
  const pendingItems = await db.select()
    .from(cardQueue)
    .where(eq(cardQueue.status, 'pending'))
    .orderBy(cardQueue.createdAt);

  console.log('[Queue] Pending items:', pendingItems.length);

  for (const item of pendingItems) {
    try {
      // Mark as processing
      await db.update(cardQueue)
        .set({ status: 'processing', attempts: item.attempts + 1, lastAttempt: new Date() })
        .where(eq(cardQueue.id, item.id));

      if (item.type === 'fund') {
        // Retry fund
        const url = new URL('https://strowallet.com/api/bitvcard/fund-withdraw-nfccard');
        url.searchParams.set('public_key', cleanKey(process.env.STROWALLET_PUBLIC_KEY!));
        url.searchParams.set('secret_key', cleanKey(process.env.STROWALLET_SECRET_KEY!));
        url.searchParams.set('card_id', item.cardId!);
        url.searchParams.set('amount', item.amount!.toString());
        url.searchParams.set('type', 'fund');
        url.searchParams.set('mode', 'live');

        const res = await fetch(url.toString(), { method: 'POST' });
        const raw = await res.json();
        console.log('[Queue] Fund retry response:', JSON.stringify(raw));

        if (raw.success) {
          // Success — mark completed
          await db.update(cardQueue)
            .set({ status: 'completed', updatedAt: new Date() })
            .where(eq(cardQueue.id, item.id));

          // Notify user
          await PushNotify.cardTopup(item.userId, item.amount);
          console.log('[Queue] Fund completed for user:', item.userId);
        } else if (raw.required) {
          // Still insufficient — back to pending
          await db.update(cardQueue)
            .set({ status: 'pending', updatedAt: new Date() })
            .where(eq(cardQueue.id, item.id));
        } else {
          // Other error — mark failed, refund user
          await db.update(cardQueue)
            .set({ status: 'failed', errorMessage: raw.message, updatedAt: new Date() })
            .where(eq(cardQueue.id, item.id));

          // Refund platform balance
          const wallet = await getUserWallet(item.userId);
          await db.update(wallets)
            .set({ availableBalance: Number(wallet.availableBalance) + Number(item.amount) })
            .where(eq(wallets.userId, item.userId));

          await PushNotify.cardFundFailed(item.userId, item.amount);
        }
      }

      if (item.type === 'create') {
        // Retry card creation
        // Re-fetch user KYC and create card
        // Similar to original create flow
        console.log('[Queue] Retrying card creation for user:', item.userId);
        // ... implementation
      }

      // Small delay between requests
      await new Promise(resolve => setTimeout(resolve, 1000));

    } catch (error) {
      console.error('[Queue] Error processing item:', item.id, error);
      await db.update(cardQueue)
        .set({ status: 'pending', updatedAt: new Date() })
        .where(eq(cardQueue.id, item.id));
    }
  }

  console.log('[Queue] Queue processing complete');
}

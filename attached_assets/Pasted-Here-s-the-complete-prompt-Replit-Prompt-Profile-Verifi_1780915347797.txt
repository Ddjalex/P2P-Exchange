-- Address verification submissions table
CREATE TABLE IF NOT EXISTS address_verifications (
  id SERIAL PRIMARY KEY,
  user_id INTEGER REFERENCES users(id) UNIQUE,
  full_name VARCHAR(255) NOT NULL,
  address_line1 VARCHAR(255) NOT NULL,
  address_line2 VARCHAR(255),
  city VARCHAR(100) NOT NULL,
  state VARCHAR(100),
  country VARCHAR(100) NOT NULL,
  postal_code VARCHAR(20),
  document_type VARCHAR(50) NOT NULL,
  -- utility_bill | bank_statement | government_letter | lease_agreement
  document_image_url TEXT NOT NULL,
  status VARCHAR(30) DEFAULT 'pending',
  -- pending | verified | rejected
  rejection_reason TEXT,
  reviewed_by INTEGER,
  submitted_at TIMESTAMP DEFAULT NOW(),
  reviewed_at TIMESTAMP
);

app.post('/api/kyc/address-verify', authenticate, async (req, res) => {
  try {
    const {
      fullName, addressLine1, addressLine2, city,
      state, country, postalCode, documentType, documentImageUrl
    } = req.body;

    // Validate required fields
    if (!fullName || !addressLine1 || !city || !country || !documentType || !documentImageUrl) {
      return res.status(400).json({ message: 'All required fields must be filled' });
    }

    // Check if already verified
    if (req.user.addressVerified) {
      return res.status(400).json({ message: 'Address already verified' });
    }

    // Upsert submission
    await db.insert(addressVerifications).values({
      userId: req.user.id,
      fullName, addressLine1,
      addressLine2: addressLine2 || null,
      city, state: state || null,
      country, postalCode: postalCode || null,
      documentType, documentImageUrl,
      status: 'pending',
      submittedAt: new Date()
    }).onConflictDoUpdate({
      target: addressVerifications.userId,
      set: {
        fullName, addressLine1,
        addressLine2: addressLine2 || null,
        city, country, documentType,
        documentImageUrl, status: 'pending',
        submittedAt: new Date()
      }
    });

    // Notify admin
    await db.insert(notifications).values({
      userId: 1, // admin
      type: 'address_submitted',
      title: '📍 New Address Verification',
      message: `${req.user.username} submitted address verification`,
      createdAt: new Date()
    });

    return res.json({ success: true, message: 'Address verification submitted successfully' });
  } catch (error) {
    console.error('Address verify error:', error);
    return res.status(500).json({ message: 'Submission failed. Please try again.' });
  }
});

// GET /api/kyc/address-status
app.get('/api/kyc/address-status', authenticate, async (req, res) => {
  try {
    const submission = await db.query.addressVerifications.findFirst({
      where: eq(addressVerifications.userId, req.user.id)
    });
    return res.json({
      status: submission?.status || null,
      submission: submission || null
    });
  } catch {
    return res.json({ status: null, submission: null });
  }
});

    const submission = await db.query.addressVerifications.findFirst({
      where: eq(addressVerifications.id, parseInt(req.params.id))
    });
    if (!submission) return res.status(404).json({ message: 'Not found' });

    // Update submission status
    await db.update(addressVerifications)
      .set({ status: 'verified', reviewedAt: new Date() })
      .where(eq(addressVerifications.id, submission.id));

    // Mark user as address verified
    await db.update(users)
      .set({ addressVerified: true, addressVerifiedAt: new Date() })
      .where(eq(users.id, submission.userId));

    // Notify user
    await db.insert(notifications).values({
      userId: submission.userId,
      type: 'address_verified',
      title: '✅ Address Verified!',
      message: 'Your address has been verified successfully.',
      createdAt: new Date()
    });

    // Telegram notification
    // TelegramNotify.addressVerified(submission.userId)

    return res.json({ success: true });
  } catch {
    return res.status(500).json({ message: 'Failed to approve' });
  }
});

app.patch('/api/admin/address-verifications/:id/reject',
  authenticateAdmin, async (req, res) => {
  try {
    const { reason } = req.body;
    const submission = await db.query.addressVerifications.findFirst({
      where: eq(addressVerifications.id, parseInt(req.params.id))
    });
    if (!submission) return res.status(404).json({ message: 'Not found' });

    await db.update(addressVerifications)
      .set({ status: 'rejected', rejectionReason: reason, reviewedAt: new Date() })
      .where(eq(addressVerifications.id, submission.id));

    await db.insert(notifications).values({
      userId: submission.userId,
      type: 'address_rejected',
      title: '❌ Address Verification Rejected',
      message: `Reason: ${reason}. Please resubmit.`,
      createdAt: new Date()
    });

    return res.json({ success: true });
  } catch {
    return res.status(500).json({ message: 'Failed to reject' });
  }
});

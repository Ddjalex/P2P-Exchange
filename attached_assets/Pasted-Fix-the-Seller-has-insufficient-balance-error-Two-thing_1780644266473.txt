}

FIX 2 — Backend: freeze wallet correctly when ad is posted
In POST /api/ads make sure sell ads freeze USDT correctly:
typescriptif (type === 'sell') {
  // Check available balance BEFORE freezing
  const wallet = await db.query.wallets.findFirst({
    where: eq(wallets.userId, req.user.id)
  });

  if (!wallet || wallet.availableBalance < totalAmount) {
    return res.status(400).json({
      message: `Insufficient USDT balance. You need ${totalAmount} USDT to post this sell ad. Current balance: ${wallet?.availableBalance ?? 0} USDT`
    });
  }

  // Freeze the USDT immediately when ad is posted
  await db.update(wallets).set({
    availableBalance: sql`available_balance - ${totalAmount}`,
    frozenBalance: sql`frozen_balance + ${totalAmount}`
  }).where(eq(wallets.userId, req.user.id));
}

// Set availableAmount = totalAmount when inserting ad

1. Card-queue-cancel: Verified refund logic
File: the route handling PUT /api/admin/cards/queue/:id/cancel
Previously, cancelling a card top-up request blindly credited the claimed amount to the user's wallet with zero verification — this was exploited to fabricate $5,510.67 in fake balance.
Fix needed: before refunding, verify a matching real deduction exists:
router.put("/cards/queue/:id/cancel", adminAuth, async (req, res) => {
  try {
    const id2 = parseInt(req.params.id);
    const item = await db.select().from(cardQueueTable).where(eq(cardQueueTable.id, id2)).then((r) => r[0]);
    if (!item) return res.status(404).json({ error: "Queue item not found" });
    if (item.status !== "pending" && item.status !== "processing") return res.status(400).json({ error: "Only pending or processing queue items can be cancelled" });
    await db.update(cardQueueTable).set({ status: "failed", errorMessage: "Cancelled by admin", updatedAt: new Date() }).where(eq(cardQueueTable.id, id2));

    let refunded = false;
    if (item.userId && item.amount) {
      try {
        const verifyResult = await db.execute(sql`
          SELECT id FROM wallet_balance_audit
          WHERE user_id = ${item.userId}
          AND (old_available - new_available) = ${parseFloat(item.amount)}
          AND changed_at > NOW() - INTERVAL '24 hours'
          AND consumed_by_queue_id IS NULL
          LIMIT 1
        `);
        if (verifyResult.rows && verifyResult.rows.length > 0) {
          const auditRowId = verifyResult.rows[0].id;
          const wallet = await db.select().from(walletsTable).where(and(eq(walletsTable.userId, item.userId), eq(walletsTable.asset, "USDT"))).then((r) => r[0]);
          if (wallet) {
            const newAvailable = (parseFloat(wallet.availableBalance) + parseFloat(item.amount)).toFixed(6);
            await db.update(walletsTable).set({ availableBalance: newAvailable, updatedAt: new Date() }).where(eq(walletsTable.id, wallet.id));
            await db.execute(sql`UPDATE wallet_balance_audit SET consumed_by_queue_id = ${id2} WHERE id = ${auditRowId}`);
            await db.insert(transactionsTable).values({
              userId: item.userId,
              type: "deposit",
              amount: item.amount,
              status: "completed",
              note: "Card fund request cancelled by admin — verified refund (matched original deduction)"
            });
            refunded = true;
          }
        }
      } catch (verifyErr) {
        console.error(`[SECURITY] Refund verification failed for queue #${id2}:`, verifyErr);
      }
    }
    await log(req.adminEmail, "cancel_card_queue", "card_queue", id2, `Cancelled card queue #${id2} for user ${item.userId}${refunded ? " (verified refund issued)" : " (NOT refunded - no matching deduction found)"}`, req);
    res.json({ success: true, refunded });
  } catch (err) {
    req.log.error({ err }, "Admin cancel card queue failed");
    res.status(500).json({ error: "Internal server error" });
  }
});

3. Card queue processor: Max retry limit
The background queue processor that retries pending "fund" items had no maximum attempt limit, causing one item to retry 40+ times over hours. Add:
const MAX_QUEUE_ATTEMPTS = 5;
// Before incrementing attempts and processing, check:
if ((item.attempts ?? 0) + 1 > MAX_QUEUE_ATTEMPTS) {
  await db.update(cardQueueTable).set({
    status: "failed",
    attempts: (item.attempts ?? 0) + 1,
    errorMessage: `Exceeded max retry attempts (${MAX_QUEUE_ATTEMPTS}) — stopped automatically for safety`,
    updatedAt: new Date()
  }).where(eq(cardQueueTable.id, item.id));
  continue; // skip to next item in the loop
}

The /api/admin/notifications/send endpoint could be triggered multiple times rapidly (e.g., a frontend double-click), sending duplicate emails/messages. Add a simple in-memory dedupe:

6. HTTP 502 → 503 for card/StroWallet errors
Cloudflare intercepts any 502 response from your origin server and replaces it with its own generic HTML error page — discarding your actual JSON error message. Please find every instance of res.status(502).json(...) related to StroWallet/card service errors (in the card creation, card funding, and merchant balance routes) and change them to res.status(503).json(...) instead. 503 is not intercepted by Cloudflare and passes your JSON through correctly.

8. P2P order freeze: Atomic race-condition fix
The freezeSellerUsdt function used a non-atomic "check balance, then update" pattern, vulnerable to a race condition where concurrent order creation could freeze more than the seller actually has. Fix:
[8/3/2026 10:06 AM] Addisu: async function freezeSellerUsdt(sellerId, amountUsdt) {
  const amount = parseFloat(amountUsdt);
  const result = await db.execute(sql`
    UPDATE wallets
    SET available_balance = (available_balance::numeric - ${amount})::text,
        frozen_balance = (frozen_balance::numeric + ${amount})::text
    WHERE user_id = ${sellerId} AND available_balance::numeric >= ${amount}
    RETURNING id
  `);
  return result.rows && result.rows.length > 0;
}
And update the order creation call site to check the return value and roll back the order if freezing fails:
if (!isBuying) {

10. Database schema (for reference — already applied directly to the production database, but should be reflected in your schema/migrations for consistency)
CREATE TABLE IF NOT EXISTS wallet_balance_audit (
  id SERIAL PRIMARY KEY,
  user_id INTEGER NOT NULL,
  old_available NUMERIC,
  new_available NUMERIC,
  old_frozen NUMERIC,
  new_frozen NUMERIC,
  changed_at TIMESTAMP NOT NULL DEFAULT NOW(),
  app_user TEXT DEFAULT current_user,
  client_addr TEXT,
  consumed_by_queue_id INTEGER
);

CREATE OR REPLACE FUNCTION log_wallet_balance_change()
RETURNS TRIGGER AS $$
BEGIN
  IF OLD.available_balance IS DISTINCT FROM NEW.available_balance
     OR OLD.frozen_balance IS DISTINCT FROM NEW.frozen_balance THEN
    INSERT INTO wallet_balance_audit (user_id, old_available, new_available, old_frozen, new_frozen, client_addr)
    VALUES (NEW.user_id, OLD.available_balance::numeric, NEW.available_balance::numeric, OLD.frozen_balance::numeric, NEW.frozen_balance::numeric, inet_client_addr()::text);
  END IF;
  RETURN NEW;
END;
$$ LANGUAGE plpgsql;

DROP TRIGGER IF EXISTS wallet_balance_change_trigger ON wallets;
CREATE TRIGGER wallet_balance_change_trigger
AFTER UPDATE ON wallets
FOR EACH ROW
EXECUTE FUNCTION log_wallet_balance_change();

CREATE TABLE IF NOT EXISTS card_queue_audit (
  id SERIAL PRIMARY KEY,
  queue_id INTEGER NOT NULL,
  user_id INTEGER,
  old_status TEXT,
  new_status TEXT,
  old_amount TEXT,
  new_amount TEXT,
  attempts INTEGER,
  error_message TEXT,
  changed_at TIMESTAMP NOT NULL DEFAULT NOW()
);

ALTER TABLE admin_logs ADD COLUMN IF NOT EXISTS ip_address TEXT;

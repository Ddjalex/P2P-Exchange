Build the Ad Share Link feature. Do not touch anything else.

HOW IT WORKS

Ad owner taps Share button on their ad card
Gets a shareable link: https://ethiop2p.com/p2p/ad/123
Sends link to friend via WhatsApp/Telegram/SMS
Friend opens link:

Has account + logged in → goes directly to order confirmation screen for that ad
Has account + not logged in → goes to login page → after login → redirects to ad order confirmation
No account → goes to register page → after register + KYC → redirects to ad order confirmation

// Share handler:
const handleShareAd = (adId: number) => {
  const shareUrl = `${window.location.origin}/p2p/ad/${adId}`;

  // Try native share (mobile)
  if (navigator.share) {
    navigator.share({
      title: 'EthioP2P Trade Offer',
      text: `Trade USDT with me on EthioP2P!`,
      url: shareUrl
    });
  } else {
    // Fallback: copy to clipboard
    navigator.clipboard.writeText(shareUrl).then(() => {
      toast.success('Link copied to clipboard!');
    });
  }
};

STEP 3 — Build src/pages/shared-ad.tsx
typescriptimport { useEffect } from 'react';
import { useParams, useLocation } from 'wouter';

export default function SharedAdPage() {
  const { adId } = useParams<{ adId: string }>();
  const [, navigate] = useLocation();
  const token = localStorage.getItem('p2p_token');

  useEffect(() => {
    // Save the intended destination
    localStorage.setItem('redirect_after_auth', `/p2p/ad/${adId}`);

    if (token) {
      // Already logged in → verify token is still valid
      fetch('/api/auth/me', {
        headers: { Authorization: `Bearer ${token}` }
      }).then(res => {
        if (res.ok) {
          // Logged in → go directly to order confirmation
          navigate(`/order-confirmation/${adId}`);
        } else {
          // Token expired → go to login
          localStorage.removeItem('p2p_token');
          navigate('/auth?redirect=' + encodeURIComponent(`/p2p/ad/${adId}`));
        }
      });
    } else {
      // Not logged in → go to auth page
      navigate('/auth?redirect=' + encodeURIComponent(`/p2p/ad/${adId}`));
    }
  }, [adId, token]);

  // Show loading while redirecting
  return (
    <div style={{
      display: 'flex',
      flexDirection: 'column',
      alignItems: 'center',
      justifyContent: 'center',
      height: '100vh',
      background: '#1a1a2e',
      gap: '16px'
    }}>
      <img src="/src/assets/logo-icon.svg" width={60} height={60} alt="EthioP2P" />
      <div style={{ color: '#00d4ff', fontSize: '14px' }}>Loading trade offer...</div>
      <div style={{
        width: '40px', height: '40px',
        border: '3px solid #334455',
        borderTop: '3px solid #00d4ff',
        borderRadius: '50%',
        animation: 'spin 1s linear infinite'
      }} />
      <style>{`@keyframes spin { to { transform: rotate(360deg); } }`}</style>
    </div>
  );
}

STEP 4 — Handle redirect after login/register
In src/pages/auth.tsx after successful login OR register:
typescript// After successful login/register:
const onAuthSuccess = () => {
  // Check if there's a pending redirect
  const redirect = localStorage.getItem('redirect_after_auth')
    || new URLSearchParams(window.location.search).get('redirect')
    || '/wallet';

  // Clear the saved redirect
  localStorage.removeItem('redirect_after_auth');

  // Navigate to destination
  navigate(redirect);
};

STEP 6 — Backend: GET /api/ads/:id (public, single ad)
typescriptapp.get('/api/ads/:id', async (req, res) => {
  try {
    const adId = parseInt(req.params.id);
    const ad = await db.query.ads.findFirst({
      where: eq(ads.id, adId)
    });

    if (!ad) {
      return res.status(404).json({ message: 'Ad not found' });
    }

    // Get advertiser info
    const advertiser = await db.query.users.findFirst({
      where: eq(users.id, ad.userId)
    });

    return res.json({
      ...ad,
      minLimit: Number(ad.minLimit) || 0,
      maxLimit: Number(ad.maxLimit) || 0,
      price: Number(ad.price) || 0,
      availableAmount: Number(ad.availableAmount) || 0,
      advertiser: {
        id: advertiser?.id,
        username: advertiser?.username,
        totalTrades: advertiser?.totalTrades ?? 0,
        completionRate: advertiser?.completionRate ?? 100,
        kycStatus: advertiser?.kycStatus
      }
    });
  } catch (error) {
    return res.status(500).json({ message: 'Failed to load ad' });
  }
});

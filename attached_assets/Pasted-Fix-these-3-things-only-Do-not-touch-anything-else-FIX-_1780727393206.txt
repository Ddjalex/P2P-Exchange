FIX 2 — Badge notifications like Instagram (red dot with count)
Create src/hooks/use-badges.ts:
typescriptimport { useQuery } from '@tanstack/react-query';

const token = () => localStorage.getItem('p2p_token');

export function useBadges() {
  // Unread messages count
  const { data: chatBadge } = useQuery({
    queryKey: ['badge-chat'],
    queryFn: () => fetch('/api/messages/unread-count', {
      headers: { Authorization: `Bearer ${token()}` }
    }).then(r => r.json()),
    refetchInterval: 5000,
    refetchIntervalInBackground: true
  });

  // New/active orders count
  const { data: orderBadge } = useQuery({
    queryKey: ['badge-orders'],
    queryFn: () => fetch('/api/orders/active-count', {
      headers: { Authorization: `Bearer ${token()}` }
    }).then(r => r.json()),
    refetchInterval: 5000,
    refetchIntervalInBackground: true
  });

  return {
    chatCount: chatBadge?.count ?? 0,
    orderCount: orderBadge?.count ?? 0,
  };
}

FIX 3 — Sound + Vibrate on new order or message
Create src/helpers/notifications.ts:
typescript// Play notification sound using Web Audio API (no file needed)
export function playNotificationSound() {
  try {
    const ctx = new (window.AudioContext || (window as any).webkitAudioContext)();

    // Two-tone notification like Instagram
    const playTone = (freq: number, start: number, duration: number, volume: number) => {
      const osc = ctx.createOscillator();
      const gain = ctx.createGain();
      osc.connect(gain);
      gain.connect(ctx.destination);
      osc.frequency.value = freq;
      osc.type = 'sine';
      gain.gain.setValueAtTime(0, ctx.currentTime + start);
      gain.gain.linearRampToValueAtTime(volume, ctx.currentTime + start + 0.01);
      gain.gain.exponentialRampToValueAtTime(0.001, ctx.currentTime + start + duration);
      osc.start(ctx.currentTime + start);
      osc.stop(ctx.currentTime + start + duration);
    };

    // Play two tones (like Instagram message sound)
    playTone(880, 0, 0.15, 0.3);
    playTone(1100, 0.18, 0.2, 0.2);
  } catch (e) {
    // Silent fail if audio not supported
  }
}

// Vibrate device
export function vibrateDevice(pattern: number[] = [100, 50, 100]) {
  try {
    if ('vibrate' in navigator) {
      navigator.vibrate(pattern);
    }
  } catch (e) {
    // Silent fail
  }
}

// Combined notification trigger
export function triggerNotification(type: 'message' | 'order') {
  if (type === 'message') {
    playNotificationSound();
    vibrateDevice([80]);             // single short vibrate
  } else if (type === 'order') {
    playNotificationSound();
    vibrateDevice([100, 50, 100]);   // double vibrate for new order
  }
}

FIX 4 — Backend badge count APIs
typescript// GET /api/messages/unread-count
app.get('/api/messages/unread-count', authenticate, async (req, res) => {
  try {
    const unread = await db.query.messages.findMany({
      where: and(
        eq(messages.receiverId, req.user.id),
        eq(messages.isRead, false),
        ne(messages.type, 'system')
      )
    });
    return res.json({ count: unread.length });
  } catch {
    return res.json({ count: 0 });
  }
});

// GET /api/orders/active-count
app.get('/api/orders/active-count', authenticate, async (req, res) => {
  try {
    const active = await db.query.orders.findMany({
      where: and(
        or(
          eq(orders.buyerId, req.user.id),
          eq(orders.sellerId, req.user.id)
        ),
        inArray(orders.status, ['unpaid', 'paid', 'appeal'])
      )
    });
    return res.json({ count: active.length });
  } catch {
    return res.json({ count: 0 });
  }
});

      .where(and(

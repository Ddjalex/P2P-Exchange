Run this SQL on Neon to fix user 110's duplicate:
sql-- Keep only the FIRST (oldest) queue item per user, cancel all duplicates
WITH ranked AS (
  SELECT id, user_id, 
         ROW_NUMBER() OVER (PARTITION BY user_id, type ORDER BY created_at ASC) as rn
  FROM card_queue
  WHERE status = 'pending' AND type = 'create'
)
UPDATE card_queue 
SET status = 'cancelled'
WHERE id IN (
  SELECT id FROM ranked WHERE rn > 1
);

-- Refund $2 for each cancelled duplicate to affected users
UPDATE wallets 
SET available_balance = available_balance + (
  SELECT COUNT(*) * 2 
  FROM card_queue cq 
  WHERE cq.user_id = wallets.user_id 
  AND cq.status = 'cancelled'
  AND cq.type = 'create'
  AND cq.created_at > NOW() - INTERVAL '1 hour'
)
WHERE user_id IN (
  SELECT DISTINCT user_id FROM card_queue 
  WHERE status = 'cancelled' 
  AND type = 'create'
  AND created_at > NOW() - INTERVAL '1 hour'

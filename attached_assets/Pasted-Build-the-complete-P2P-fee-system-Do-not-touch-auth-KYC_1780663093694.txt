Build the complete P2P fee system. Do not touch auth, KYC, or admin dashboard layout.

FEE MODEL OVERVIEW
Maker fee = charged to AD POSTER when order completes
Taker fee = charged to ORDER PLACER when order completes
Both fees deducted from USDT amount received
Fee goes to platform fee wallet
Default: Maker 0.20% + Taker 0.10%
Admin can change % anytime from fee management panel

PART 2 — FEE CALCULATION HELPER
Create artifacts/api-server/src/helpers/fees.ts:
typescriptimport { db } from '../db';
import { feeSettings } from '../schema';
import { eq } from 'drizzle-orm';

export async function getFeePercents() {
  const fees = await db.query.feeSettings.findMany();
  const feeMap: Record<string, number> = {};
  fees.forEach(f => { feeMap[f.feeType] = Number(f.value); });
  return {
    makerFeePercent: feeMap['maker_fee_percent'] ?? 0.20,
    takerFeePercent: feeMap['taker_fee_percent'] ?? 0.10,
    withdrawalFeeTRC20: feeMap['withdrawal_fee_trc20'] ?? 1.00,
    withdrawalFeeERC20: feeMap['withdrawal_fee_erc20'] ?? 5.00,
  };
}

export function calculateFees(grossUsdt: number, makerPercent: number, takerPercent: number) {
  const makerFee = parseFloat((grossUsdt * makerPercent / 100).toFixed(8));
  const takerFee = parseFloat((grossUsdt * takerPercent / 100).toFixed(8));
  const totalFee = makerFee + takerFee;
  const netUsdt = parseFloat((grossUsdt - totalFee).toFixed(8));
  return { makerFee, takerFee, totalFee, netUsdt };
}

PART 3 — APPLY FEES ON ORDER RELEASE
In PATCH /api/orders/:id/release update the release logic:
typescriptimport { getFeePercents, calculateFees } from '../helpers/fees';

app.patch('/api/orders/:id/release', authenticate, async (req, res) => {
  try {
    const orderId = parseInt(req.params.id);
    const order = await db.query.orders.findFirst({
      where: eq(orders.id, orderId)
    });

    if (!order) return res.status(404).json({ message: 'Order not found' });
    if (order.sellerId !== req.user.id) return res.status(403).json({ message: 'Not authorized' });
    if (order.status !== 'paid') return res.status(400).json({ message: 'Order is not in paid status' });

    // Get current fee percentages
    const { makerFeePercent, takerFeePercent } = await getFeePercents();

    // Calculate fees
    const grossUsdt = Number(order.amountUsdt);
    const { makerFee, takerFee, totalFee, netUsdt } = calculateFees(
      grossUsdt,
      makerFeePercent,
      takerFeePercent
    );

    // Identify maker and taker
    // Maker = ad poster (could be buyer or seller depending on ad type)
    const ad = await db.query.ads.findFirst({ where: eq(ads.id, order.adId) });
    const makerId = ad.userId;           // ad poster = maker
    const takerId = makerId === order.buyerId
      ? order.sellerId
      : order.buyerId;                   // other party = taker

    // Buyer receives USDT = gross - both fees
    const buyerReceives = netUsdt;

    await db.transaction(async (tx) => {
      // 1. Release seller frozen USDT
      await tx.update(wallets)
        .set({ frozenBalance: sql`frozen_balance - ${grossUsdt}` })
        .where(eq(wallets.userId, order.sellerId));

      // 2. Credit buyer with NET amount (after fees)
      await tx.update(wallets)
        .set({ availableBalance: sql`available_balance + ${buyerReceives}` })
        .where(eq(wallets.userId, order.buyerId));

      // 3. Add fees to platform wallet
      await tx.update(platformWallet)
        .set({ totalCollected: sql`total_collected + ${totalFee}`, updatedAt: new Date() })
        .where(eq(platformWallet.asset, 'USDT'));

      // 4. Update order with fee details
      await tx.update(orders)
        .set({
          status: 'completed',
          makerFeePercent,
          takerFeePercent,
          makerFeeAmount: makerFee,
          takerFeeAmount: takerFee,
          makerNetAmount: grossUsdt - makerFee,
          takerNetAmount: buyerReceives,
          releasedAt: new Date(),
          completedAt: new Date()
        })
        .where(eq(orders.id, orderId));

      // 5. Log fee transactions
      await tx.insert(feeTransactions).values([
        {
          orderId,
          userId: makerId,
          feeType: 'maker',
          feePercent: makerFeePercent,
          grossAmount: grossUsdt,
          feeAmount: makerFee,
          netAmount: grossUsdt - makerFee,
          createdAt: new Date()
        },
        {
          orderId,
          userId: takerId,
          feeType: 'taker',
          feePercent: takerFeePercent,
          grossAmount: grossUsdt,
          feeAmount: takerFee,
          netAmount: buyerReceives,
          createdAt: new Date()
        }
      ]);

      // 6. Transaction records
      await tx.insert(transactions).values([
        {
          userId: order.buyerId,
          type: 'p2p_buy',
          amount: buyerReceives,
          status: 'completed',
          note: `Bought ${grossUsdt} USDT. Fee: ${totalFee.toFixed(8)} USDT. Received: ${buyerReceives} USDT`,
          createdAt: new Date()
        },
        {
          userId: order.sellerId,
          type: 'p2p_sell',
          amount: grossUsdt,
          status: 'completed',
          note: `Sold ${grossUsdt} USDT`,
          createdAt: new Date()
        }
      ]);

      // 7. Notifications
      await tx.insert(notifications).values([
        {
          userId: order.buyerId,
          type: 'order_completed',
          title: '✅ Order Completed',
          message: `You received ${buyerReceives.toFixed(4)} USDT (fee: ${totalFee.toFixed(4)} USDT deducted)`,
          relatedOrderId: orderId,
          createdAt: new Date()
        },
        {
          userId: order.sellerId,
          type: 'order_completed',
          title: '✅ Order Completed',
          message: `Order #${order.orderNo} completed. Br ${Number(order.amountEtb).toLocaleString()} received.`,
          relatedOrderId: orderId,
          createdAt: new Date()
        }
      ]);
    });

    return res.json({ message: 'Crypto released successfully' });

  } catch (error) {
    console.error('Release error:', error);
    return res.status(500).json({ message: 'Failed to release crypto' });
  }
});

function FeeRow({ key, label, unit, description, currentValue, onSave }) {
  const [editing, setEditing] = useState(false);
  const [value, setValue] = useState(currentValue);

  return (
    <div style={{ display: 'flex', justifyContent: 'space-between',
      alignItems: 'center', padding: '14px 0',
      borderBottom: '1px solid #1e2d3d' }}>
      <div>
        <div style={{ color: '#fff', fontSize: '13px', fontWeight: 600 }}>{label}</div>
        <div style={{ color: '#556677', fontSize: '11px' }}>{description}</div>
      </div>
      <div style={{ display: 'flex', alignItems: 'center', gap: '8px' }}>
        {editing ? (
          <>
            <input
              type="number"
              value={value}
              onChange={e => setValue(e.target.value)}
              step="0.01"
              min="0"
              max="10"
              style={{
                width: '80px', background: '#1e2d3d',
                border: '1px solid #00d4ff', borderRadius: '6px',
                padding: '6px 8px', color: '#fff',
                fontSize: '13px', outline: 'none'
              }}
            />
            <span style={{ color: '#8899aa', fontSize: '12px' }}>{unit}</span>
            <button onClick={() => { onSave(key, parseFloat(value)); setEditing(false); }}
              style={{ background: '#00d4ff', border: 'none', borderRadius: '6px',
                padding: '6px 12px', color: '#1a1a2e', fontSize: '12px',
                fontWeight: 700, cursor: 'pointer' }}>Save</button>
            <button onClick={() => setEditing(false)}
              style={{ background: 'none', border: '1px solid #334455',
                borderRadius: '6px', padding: '6px 10px',
                color: '#8899aa', fontSize: '12px', cursor: 'pointer' }}>Cancel</button>
          </>
        ) : (
          <>
            <span style={{ color: '#00d4ff', fontSize: '16px', fontWeight: 700 }}>
              {currentValue}{unit}
            </span>
            <button onClick={() => { setValue(currentValue); setEditing(true); }}
              style={{ background: 'none', border: '1px solid #334455',

          </>
        )}

  );
}

    const { feeType, value } = req.body;

    });

    return res.status(500).json({ message: 'Failed to update fee' });
  }
});

FEE FLOW SUMMARY
Order created:
  Gross USDT = ETB ÷ price
  Maker fee = Gross × 0.20%
  Taker fee = Gross × 0.10%
  Total fee = 0.30%
  Net USDT to buyer = Gross - Total fee

Example:
  Buy 100 USDT at 178 ETB
  Gross = 100 USDT
  Maker fee = 0.20 USDT
  Taker fee = 0.10 USDT
  Buyer receives = 99.70 USDT
  Platform earns = 0.30 USDT

Admin can change % anytime → takes effect on next order
Fee history logged in fee_transactions table
Platform total shown in admin fee management page

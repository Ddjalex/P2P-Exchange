Fix: Block duplicate card creation — one pending request per user maximum
Problem: User can click "Create My Card" multiple times creating duplicate queue entries. Each attempt deducts $2 fee separately.
Fix 1 — Backend check in POST /api/cards/create:
Add this check at the VERY START before any fee deduction:
typescript// Check 1: User already has a card
const existingCard = await db.select()
  .from(cardsTable)
  .where(eq(cardsTable.userId, userId))
  .limit(1);

if (existingCard.length > 0) {
  return res.status(400).json({ 
    error: 'You already have a card.' 
  });
}

// Check 2: User already has a pending queue item
const existingQueue = await db.select()
  .from(cardQueueTable)
  .where(
    and(
      eq(cardQueueTable.userId, userId),
      eq(cardQueueTable.type, 'create'),
      inArray(cardQueueTable.status, ['pending', 'processing'])
    )
  );

if (existingQueue.length > 0) {
  return res.status(400).json({
    success: true,
    queued: true,
    message: 'Your card creation is already queued. Please wait — you will be notified when your card is ready. No additional charges will apply.'
  });
}

// ONLY THEN proceed with fee deduction and StroWallet call
Fix 2 — Frontend: disable button after first click:
In card.tsx card creation confirmation screen:
tsxconst [creating, setCreating] = useState(false);
const [queued, setQueued] = useState(false);

const handleCreateCard = async () => {
  if (creating || queued) return;
  setCreating(true);
  
  try {
    const result = await createCard(country);
    
    if (result.queued) {
      setQueued(true); // permanently disable button
      toast.success('⏳ Card creation queued! You will be notified when ready.');
    } else {
      toast.success('🎉 Card created successfully!');
    }
  } catch (e) {
    toast.error(e.message);
    setCreating(false); // re-enable on error only
  }
  // Don't re-enable if success or queued
};

<button
  onClick={handleCreateCard}
  disabled={creating || queued}
  className={creating ? 'btn-loading' : queued ? 'btn-queued' : 'btn-primary'}
>
  {creating ? '⏳ Creating...' : queued ? '✅ Request Queued' : 'Create My Card'}
</button>
